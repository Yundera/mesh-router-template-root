# Mesh-router self-check scripts — executed in this order by self-check.sh.
# No auto-discovery: a script only runs if listed here.

# Self-maintenance
ensure-scripts-executable.sh
ensure-nightly-self-check.sh
ensure-logrotate.sh

# Prerequisites
ensure-docker-installed.sh
# yq, for tools/authelia-user-manager.sh. Non-fatal if it cannot be installed —
# nothing in the boot path needs it (ensure-dex.sh's claimed-ness probe fails
# open without it); only the interactive user-management verbs do.
ensure-yq-installed.sh
ensure-env-valid.sh

# Template sync / auto-update (no-op when MESH_AUTO_UPDATE=false).
# Also where scripts/migrations/ runs, from the newly downloaded tree.
ensure-template-sync.sh

# Stack
ensure-public-ip.sh
ensure-email-synced.sh
# ensure-authelia.sh MUST run before ensure-connector-local-account.sh: it mints
# AUTHELIA_DEX_SECRET into .env, which that script writes into the Dex "Local
# Account" connector's clientSecret. All of them run BEFORE ensure-auth-stack.sh
# so each service finds its config on first start.
ensure-authelia.sh
# Dex connectors are drop-ins, one ensure-connector-* script each, all BEFORE
# ensure-dex.sh: it appends whatever connectors.d/ holds and, with nothing there,
# keeps Dex absent (it cannot start on an empty list).
ensure-connector-local-account.sh
# ensure-dex-session-key.sh MUST run before ensure-dex.sh: it mints
# DEX_SESSION_KEY into .env, which ensure-dex.sh interpolates into
# `sessions.cookieEncryptionKey`. Unset renders an empty key, which Dex accepts
# (cookies simply go unencrypted), so this is an ordering requirement rather than
# a hard dependency.
ensure-dex-session-key.sh
ensure-dex.sh
ensure-stack-pulled.sh
# ensure-stack-up.sh also creates the `pcs` network every stack joins as external.
ensure-stack-up.sh
# The auth stack (dex, authelia, auth-registrar) — split out of the mesh stack.
# Dex only when ensure-dex.sh rendered a connector (dex_wanted).
# MUST stay after ensure-stack-up.sh (Caddy routes auth-* and local-auth-*, smtp
# relays Authelia's mail) and after the renders above. Its first run is the
# cutover: it evicts the mesh-project copies of those containers and recreates
# them here (see the header of ensure-auth-stack.sh).
ensure-auth-stack.sh

# Auxiliary stack — Maison, the dashboard. MUST stay after ensure-auth-stack.sh:
# its AppShield gate needs auth-registrar / dex reachable by name on `pcs`.
ensure-maison-stack.sh
# Retired, and listed here only so nobody reintroduces it: ensure-maison-app-mirror.sh.
# It projected an app's compose + a generated .env from /DATA/AppData/casaos/apps/<app>
# into ${DATA_ROOT}/AppData/<app>. CasaOS stopped writing that tree when it was removed
# (2026-08-02), so the copy has been complete ever since; Maison owns that path directly
# now and re-applies the deployment's variables into each app's .env on every start
# (Maison's docs/app-env.md), so a second writer of that file would remove the keys
# Maison computes. Whatever is left under /DATA/AppData/casaos/apps is dead data.

# The web Terminal - the AppStore's Terminal app shipped as a system app
# (stacks/terminal/). Same constraint as Maison above: its gate registers through
# auth-registrar.
# Opt out with TERMINAL_ENABLED=false in the mesh .env.
ensure-terminal-stack.sh

# Verification (check-only, no repair)
# ensure-root-domain.sh MUST stay after ensure-maison-stack.sh: the root domain's
# default target is Maison, so probing before it is up reports a failure that is only
# a race. Covers the one surface no other step looks at — ${DOMAIN} and the catch-all
# come from the Caddyfile's DEFAULT_SERVICE_HOST, not from any container label, so a
# stale value there passes every other check while the front door 502s.
ensure-root-domain.sh
ensure-route-registered.sh
